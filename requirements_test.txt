# Test-only dependencies for tests/ha/** (real-Home-Assistant integration tests).
#
# Install into a dedicated venv (Python >= 3.14, matching live HA Core 2026.9.x):
#   uv venv .venv-ha --python 3.14
#   uv pip install --python .venv-ha/bin/python -r requirements_test.txt
#
# On this host `/data` runs out of space for the HA frontend wheel (>500 MB unpacked), so the
# venv lives on the roomier root filesystem and `.venv-ha` is a symlink to it:
#   export UV_CACHE_DIR=/var/tmp/uv-cache-iledclock
#   uv venv /var/tmp/iledclock-venv-ha --python 3.14
#   ln -sfn /var/tmp/iledclock-venv-ha .venv-ha
#   uv pip install --python .venv-ha/bin/python -r requirements_test.txt
# Either way the acceptance command is `.venv-ha/bin/pytest tests/ha -q`.
#
# pytest-homeassistant-custom-component pins an exact `homeassistant` release (and, with
# it, pytest/pytest-asyncio/syrupy versions known to work together) -- do not loosen it.
pytest-homeassistant-custom-component==0.13.366
