# Test-only dependencies for tests/ha/** (real-Home-Assistant integration tests).
#
# Install into a dedicated venv (Python >= 3.14, as Home Assistant Core 2026.11 requires):
#   uv venv .venv-ha --python 3.14
#   uv pip install --python .venv-ha/bin/python -r requirements_test.txt
#
# On this host `/data` runs out of space for the HA frontend wheel (>500 MB unpacked), so the
# venv lives on the roomier root filesystem and `.venv-ha` is a symlink to it:
#   export UV_CACHE_DIR=/var/tmp/uv-cache-iledclock
#   uv venv /var/tmp/iledclock-venv-ha --python 3.14
#   ln -sfn /var/tmp/iledclock-venv-ha .venv-ha
#   uv pip install --python .venv-ha/bin/python -r requirements_test.txt
# Either way the acceptance command is `.venv-ha/bin/pytest tests/ha -q`.
#
# pytest-homeassistant is the test harness Home Assistant itself ships (it replaces the
# discontinued pytest-homeassistant-custom-component). It is versioned exactly like Home
# Assistant and pins that exact `homeassistant` release (and, with it, the pytest, pytest-asyncio
# and syrupy versions known to work together) -- do not loosen it. The `bluetooth` extra brings
# the Bluetooth stack the integration's `bluetooth_adapters` -> `bluetooth` -> `usb` dependency
# chain imports (bleak-retry-connector, bluetooth-adapters, habluetooth, aiousbwatcher, ...).
# To test another Home Assistant release, change this one line and the frontend pin below.
pytest-homeassistant[bluetooth]==2026.11.0

# --- HA component requirements reached through this integration's manifest "dependencies" ---
# custom_components/iledclock/manifest.json declares no "requirements" of its own, but its
# dependencies do, and HA normally pip-installs those lazily on first use. A test run must not do
# that (it would hit the network mid-test), so the one the harness does not bring is pinned here
# to the exact version homeassistant/components/frontend/manifest.json requests for HA 2026.11.0.
# `frontend`: __init__.py's `add_extra_js_url` / `_async_register_frontend` needs it importable,
# and its own `hass_frontend` package provides the panel's static assets.
home-assistant-frontend==20261028.4

# --- gallery Divoom decoder requirements ---
# custom_components/iledclock/gallery/__init__.py's `DIVOOM_REQUIREMENTS`: only needed to
# decode real Divoom Cloud "pixel bean" media (AES/LZO/zstd containers); pre-installed here
# so `_ensure_divoom_requirements` never tries a runtime pip install during a test. Left
# unpinned exactly as the integration itself requests them.
pycryptodome
lzallright
zstandard
